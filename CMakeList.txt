cmake_minimum_required(VERSION 3.10)
project(corporate_portal_backend)

# Поиск библиотеки libpqxx
find_package(libpqxx REQUIRED)

# Добавление исходных файлов
add_executable(corporate_portal_backend src/main.cpp src/database.cpp)

# Подключение библиотеки libpqxx
target_link_libraries(corporate_portal_backend PRIVATE libpqxx::pqxx)
